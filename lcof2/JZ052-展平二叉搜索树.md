## Code

1. 创建一个虚拟头节点 `dummy`，用于简化边界情况的处理。
2. 使用中序遍历（左-根-右）遍历二叉搜索树，这样可以按升序访问所有节点。
3. 对于每个访问到的节点，将其连接到当前指针 `curr` 的右子节点。
4. 清空当前节点的左子节点，并更新 `curr` 指针为当前节点。
5. 最终返回 `dummy->right` 作为新树的根节点。

时间复杂度：$O(n)$，其中 $n$ 是二叉树的节点数，需要遍历所有节点。空间复杂度：$O(h)$，其中 $h$ 是树的高度，递归调用栈的深度。

```cpp
class Solution {
private:
    TreeNode* curr;
    
    void inorder(TreeNode* root) {
        if (root == nullptr) return;
        
        // Inorder traversal: left -> root -> right
        inorder(root->left);
        
        // Connect current node to the chain
        curr->right = root;
        root->left = nullptr;
        curr = root;
        
        inorder(root->right);
    }
    
public:
    TreeNode* increasingBST(TreeNode* root) {
        TreeNode* dummy = new TreeNode(0);
        curr = dummy;
        
        inorder(root);
        
        return dummy->right;
    }
};
```
/**
 * Definition for a binary tree node.
 * struct TreeNode {
 *     int val;
 *     TreeNode *left;
 *     TreeNode *right;
 *     TreeNode() : val(0), left(nullptr), right(nullptr) {}
 *     TreeNode(int x) : val(x), left(nullptr), right(nullptr) {}
 *     TreeNode(int x, TreeNode *left, TreeNode *right) : val(x), left(left), right(right) {}
 * };
 */
class Solution {
public:
    TreeNode* increasingBST(TreeNode* root) {
        TreeNode* dummy = new TreeNode(-1);
        TreeNode* curr = dummy;
        inorder(root, curr);
        return dummy->right;
    }

private:
    void inorder(TreeNode* node, TreeNode*& curr) {
        if (!node) {
            return;
        }
        inorder(node->left, curr);
        curr->right = node;
        node->left = nullptr;
        curr = node;
        inorder(node->right, curr);
    }
};
```
